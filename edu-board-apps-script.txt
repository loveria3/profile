function doGet(e) {

}

// ──────────────────────────────────────────────────────────
//  doPost — 목록(list) / 추가(add) / 수정(update) / 삭제(delete)
//  모든 요청에 pin 필요
// ──────────────────────────────────────────────────────────
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);

    var pinErr = checkPin_(data.pin);
    if (pinErr) return errorOutput(pinErr, 'pin');

    var sheet = getSheet_();

    if (data.action === 'list') {
      return jsonOutput({ success: true, items: readItems_(sheet), lastCollect: lastCollect_() });
    }

    if (data.action === 'delete') {
      var delIdx = findRowById_(sheet, data.id);
      if (delIdx < 0) return errorOutput('해당 공고를 찾을 수 없습니다: ' + data.id);
      sheet.deleteRow(delIdx);
      return jsonOutput({ success: true });
    }

    // 지금 수집 (페이지의 '지금 수집' 버튼)
    if (data.action === 'collect') {
      var r = runCollect_(sheet);
      return jsonOutput({
        success: true, added: r.added, scanned: r.scanned, purged: r.purged,
        items: readItems_(sheet), lastCollect: lastCollect_()
      });
    }

    var item = data.item || {};
    if (!String(item.title || '').trim()) return errorOutput('제목은 꼭 적어야 합니다.');

    // 수정 (등록일·ID·출처 유지)
    if (data.action === 'update') {
      var rowIdx = findRowById_(sheet, data.id);
      if (rowIdx < 0) return errorOutput('해당 공고를 찾을 수 없습니다: ' + data.id);
      var old = sheet.getRange(rowIdx, 1, 1, HEADER.length).getValues()[0];
      writeRow_(sheet, rowIdx, toRow_(item, old[9], old[10], old[11]));
      return jsonOutput({ success: true });
    }

    // 추가
    if (data.action === 'add') {
      var newId = genId_();
      writeRow_(sheet, sheet.getLastRow() + 1, toRow_(item, today_(), newId, SOURCE_MANUAL));
      return jsonOutput({ success: true, id: newId });
    }

    return errorOutput('알 수 없는 요청입니다: ' + data.action);
  } catch (err) {
    return errorOutput(err.toString());
  }
}

// ──────────────────────────────────────────────────────────
//  PIN 확인 (스크립트 속성 EDU_PIN) + 연속 실패 잠금
// ──────────────────────────────────────────────────────────
function checkPin_(pin) {
  var expected = PropertiesService.getScriptProperties().getProperty('EDU_PIN');
  if (!expected) return '스크립트 속성 EDU_PIN 이 설정되지 않았습니다. (배포 방법 3번 참고)';

  var cache = CacheService.getScriptCache();
  var fails = Number(cache.get('edu_pin_fail') || 0);
  if (fails >= MAX_FAIL) return 'PIN 을 여러 번 틀려 10분 동안 잠겼습니다.';

  if (String(pin || '') !== String(expected)) {
    cache.put('edu_pin_fail', String(fails + 1), LOCK_SEC);
    return 'PIN 이 올바르지 않습니다.';
  }
  if (fails) cache.remove('edu_pin_fail');
  return '';
}

// ──────────────────────────────────────────────────────────
//  내부 헬퍼
// ──────────────────────────────────────────────────────────
function getSheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow(HEADER);
    sheet.setFrozenRows(1);
    sheet.getRange('D:D').setNumberFormat('@'); // 마감일을 글자 그대로 저장
    return sheet;
  }
  // v1 로 만든 시트에는 L열(출처)이 없다 — 한 번만 넓혀 준다
  if (sheet.getMaxColumns() < HEADER.length) {
    sheet.insertColumnsAfter(sheet.getMaxColumns(), HEADER.length - sheet.getMaxColumns());
  }
  if (!String(sheet.getRange(1, HEADER.length).getValue() || '').trim()) {
    sheet.getRange(1, HEADER.length).setValue(HEADER[HEADER.length - 1]);
  }
  return sheet;
}

function readItems_(sheet) {
  var data = sheet.getDataRange().getValues();
  var list = [];
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    var title = String(row[1] || '').trim();
    if (!title) continue;
    var id = String(row[10] || '').trim();
    if (!id) { id = genId_(); try { sheet.getRange(i + 1, 11).setValue(id); } catch (e) {} }
    list.push({
      type:     String(row[0] || '').trim(),
      title:    title,
      org:      String(row[2] || '').trim(),
      deadline: dateStr_(row[3]),
      period:   String(row[4] || '').trim(),
      place:    String(row[5] || '').trim(),
      link:     String(row[6] || '').trim(),
      status:   String(row[7] || '').trim(),
      memo:     String(row[8] || '').trim(),
      date:     dateStr_(row[9]),
      id:       id,
      source:   String(row[11] || '').trim() || SOURCE_MANUAL
    });
  }
  return list;
}

function toRow_(p, date, id, source) {
  var type   = String(p.type || '').trim();
  var status = String(p.status || '').trim();
  var dl     = String(p.deadline || '').trim();
  var src    = String(source || p.source || '').trim();
  return [
    TYPES.indexOf(type) >= 0 ? type : TYPES[0],
    String(p.title || '').trim(),
    String(p.org || '').trim(),
    /^\d{4}-\d{2}-\d{2}$/.test(dl) ? dl : '',
    String(p.period || '').trim(),
    String(p.place || '').trim(),
    String(p.link || '').trim(),
    ALL_STATUSES.indexOf(status) >= 0 ? status : STATUSES[0],
    String(p.memo || '').trim(),
    date,
    id,
    src || SOURCE_MANUAL
  ];
}

function writeRow_(sheet, rowIdx, row) {
  sheet.getRange(rowIdx, 4).setNumberFormat('@');
  sheet.getRange(rowIdx, 1, 1, HEADER.length).setValues([row]);
}

function dateStr_(d) {
  return (d instanceof Date)
    ? Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd')
    : String(d || '').trim();
}

function today_() {
  return Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd');
}

function genId_() {
  return Date.now().toString(36) + Math.random().toString(36).slice(2, 6);
}

function findRowById_(sheet, id) {
  var rows = sheet.getDataRange().getValues();
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][10]).trim() === String(id).trim()) return i + 1;
  }
  return -1;
}

function jsonOutput(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function errorOutput(msg, code) {
  return ContentService
    .createTextOutput(JSON.stringify({ error: true, code: code || '', message: msg }))
    .setMimeType(ContentService.MimeType.JSON);
}

// 매일 오전 7시 자동 실행 (setupCollectTrigger 로 설치)
function collectPostings() {
  var r = runCollect_(getSheet_());
  Logger.log('자동 수집: 훑은 기사 ' + r.scanned + '건 / 새로 담음 ' + r.added + '건 / 정리 ' + r.purged + '건');
  return r;
}

// 트리거 설치 — 편집기에서 한 번만 ▶실행 하세요
function setupCollectTrigger() {
  var all = ScriptApp.getProjectTriggers();
  for (var i = 0; i < all.length; i++) {
    if (all[i].getHandlerFunction() === 'collectPostings') ScriptApp.deleteTrigger(all[i]);
  }
  ScriptApp.newTrigger('collectPostings').timeBased().atHour(7).everyDays(1).create();
  var msg = '매일 오전 7시 자동 수집 트리거를 설치했습니다.';
  Logger.log(msg);
  return msg;
}

function runCollect_(sheet) {
  var purged = purgeOldReview_(sheet);

  // 이미 있는 제목은 다시 담지 않는다
  var seen = {};
  var existing = readItems_(sheet);
  for (var i = 0; i < existing.length; i++) seen[normKey_(existing[i].title)] = true;

  var feeds = fetchFeeds_(COLLECT_KEYWORDS);
  var added = 0, scanned = 0;
  var rows = [];

  for (var k = 0; k < COLLECT_KEYWORDS.length; k++) {
    var kw = COLLECT_KEYWORDS[k];
    var entries = feeds[k] || [];
    scanned += entries.length;
    var taken = 0;

    for (var j = 0; j < entries.length; j++) {
      if (taken >= COLLECT_MAX_PER_KW || added >= COLLECT_MAX_TOTAL) break;
      var e = entries[j];
      var title = stripSource_(e.title);
      if (!title || !passFilter_(title)) continue;

      var key = normKey_(title);
      if (seen[key]) continue;
      seen[key] = true;

      rows.push(toRow_({
        type:     classify_(title, kw.type),
        title:    title,
        org:      e.source,
        deadline: findDeadline_(title + ' ' + e.desc),
        link:     e.link,
        status:   REVIEW_STATUS,
        memo:     '자동 수집 · 검색어 「' + kw.q + '」'
      }, today_(), genId_(), SOURCE_AUTO));

      added++; taken++;
    }
    if (added >= COLLECT_MAX_TOTAL) break;
  }

  if (rows.length) {
    var start = sheet.getLastRow() + 1;
    sheet.getRange(start, 4, rows.length, 1).setNumberFormat('@');
    sheet.getRange(start, 1, rows.length, HEADER.length).setValues(rows);
  }
  PropertiesService.getScriptProperties().setProperty('EDU_LAST_COLLECT', today_() + ' ' +
    Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'HH:mm'));

  return { added: added, scanned: scanned, purged: purged };
}

// 검색어 전부를 한 번에 받아온다 (fetchAll 이라 하나씩 받는 것보다 훨씬 빠르다)
function fetchFeeds_(keywords) {
  var reqs = [];
  for (var i = 0; i < keywords.length; i++) {
    reqs.push({
      url: 'https://news.google.com/rss/search?q='
         + encodeURIComponent(keywords[i].q + ' when:' + COLLECT_WINDOW)
         + '&hl=ko&gl=KR&ceid=KR:ko',
      muteHttpExceptions: true,
      followRedirects: true
    });
  }
  var res = [];
  try {
    res = UrlFetchApp.fetchAll(reqs);
  } catch (err) {
    Logger.log('RSS 수신 실패: ' + err);
    return [];
  }

  var out = [];
  for (var j = 0; j < res.length; j++) out.push(parseRss_(res[j]));
  return out;
}

function parseRss_(res) {
  var list = [];
  try {
    if (!res || res.getResponseCode() !== 200) return list;
    var channel = XmlService.parse(res.getContentText()).getRootElement().getChild('channel');
    if (!channel) return list;
    var items = channel.getChildren('item');
    for (var i = 0; i < items.length; i++) {
      var it = items[i];
      var src = it.getChild('source');
      list.push({
        title:  childText_(it, 'title'),
        link:   childText_(it, 'link'),
        desc:   childText_(it, 'description'),
        source: src ? String(src.getText()).trim() : ''
      });
    }
  } catch (err) {
    Logger.log('RSS 해석 실패: ' + err);
  }
  return list;
}

function childText_(el, name) {
  var c = el.getChild(name);
  return c ? String(c.getText()).trim() : '';
}

// 구글 뉴스 제목은 "제목 - 언론사" 꼴이라 뒤의 언론사를 뗀다
function stripSource_(t) {
  return String(t || '').replace(/\s+-\s+[^\-]+$/, '').trim();
}

// 중복 판정용 — 기호·공백을 지우고 앞 20글자만 본다
function normKey_(t) {
  return String(t || '').toLowerCase().replace(/[^0-9a-z가-힣]/g, '').slice(0, 20);
}

// 버릴 말이 없고 + 모집을 뜻하는 말이 있고 + 내 분야를 뜻하는 말이 있어야 통과
function passFilter_(title) {
  for (var i = 0; i < DROP_WORDS.length; i++) {
    if (title.indexOf(DROP_WORDS[i]) >= 0) return false;
  }
  if (!hasAny_(title, MUST_WORDS)) return false;
  if (!hasAny_(title, TOPIC_WORDS)) return false;
  return true;
}

function hasAny_(title, words) {
  for (var i = 0; i < words.length; i++) {
    if (title.indexOf(words[i]) >= 0) return true;
  }
  return false;
}

function classify_(title, fallback) {
  var isEdu = false, isJob = false;
  for (var i = 0; i < EDU_WORDS.length; i++) if (title.indexOf(EDU_WORDS[i]) >= 0) { isEdu = true; break; }
  for (var j = 0; j < JOB_WORDS.length; j++) if (title.indexOf(JOB_WORDS[j]) >= 0) { isJob = true; break; }
  if (isJob && !isEdu) return '강사모집';
  if (isEdu && !isJob) return '연수·교육';
  return TYPES.indexOf(fallback) >= 0 ? fallback : TYPES[0];
}

// 마감일 추출 — '까지'·'마감'·'~' 같은 말이 붙어 있을 때만 날짜로 본다.
// (틀린 날짜를 넣으면 멀쩡한 공고가 지난 것처럼 보이므로, 애매하면 비워 둔다)
function findDeadline_(text) {
  var s = String(text || '').replace(/<[^>]*>/g, ' ').replace(/&[a-z#0-9]+;/gi, ' ');

  var full = s.match(/(20\d{2})\s*[.\-\/년]\s*(\d{1,2})\s*[.\-\/월]\s*(\d{1,2})\s*일?\s*(?:까지|마감)/);
  if (full) return ymd_(Number(full[1]), Number(full[2]), Number(full[3]));

  var pats = [
    /(\d{1,2})\s*월\s*(\d{1,2})\s*일\s*(?:까지|마감)/,
    /마감[^0-9]{0,8}(\d{1,2})\s*[.\/월]\s*(\d{1,2})/,
    /~\s*(\d{1,2})\s*[.\/월]\s*(\d{1,2})\s*일?/
  ];
  for (var i = 0; i < pats.length; i++) {
    var m = s.match(pats[i]);
    if (!m) continue;
    var mo = Number(m[1]), da = Number(m[2]);
    if (mo < 1 || mo > 12 || da < 1 || da > 31) continue;
    var now = new Date();
    var y = now.getFullYear();
    // 사흘 넘게 지난 날짜면 내년 것으로 본다 (연말에 내년 공고가 올라오는 경우)
    if (new Date(y, mo - 1, da).getTime() < now.getTime() - 3 * 86400000) y += 1;
    return ymd_(y, mo, da);
  }
  return '';
}

function ymd_(y, m, d) {
  if (!(m >= 1 && m <= 12 && d >= 1 && d <= 31)) return '';
  return y + '-' + ('0' + m).slice(-2) + '-' + ('0' + d).slice(-2);
}

// 오래 방치된 '검토대기' 를 치운다 (아래에서 위로 지워야 줄번호가 안 밀린다)
function purgeOldReview_(sheet) {
  var rows = sheet.getDataRange().getValues();
  var limit = new Date().getTime() - REVIEW_KEEP_DAYS * 86400000;
  var removed = 0;
  for (var i = rows.length - 1; i >= 1; i--) {
    if (String(rows[i][7] || '').trim() !== REVIEW_STATUS) continue;
    var d = dateStr_(rows[i][9]);
    if (!/^\d{4}-\d{2}-\d{2}$/.test(d)) continue;
    if (new Date(d + 'T00:00:00').getTime() >= limit) continue;
    sheet.deleteRow(i + 1);
    removed++;
  }
  return removed;
}

}
