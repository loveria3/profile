/* 강의스케쥴 시트 O열 = 참여형태 (강사 / 보조강사 / 안전요원)
   ⚠️ 컬럼 번호가 파일마다 흩어져 있으니 새 컬럼을 만들 때 반드시 시트를 열어

     O(15) 참여형태    → 여기
     P(16) 예상금액    → 여기 (노션 '예정' 행 생성용)
     Q(17) 요청ID      → 여기 (같은 저장이 두 번 들어가는 것 방지)        */
var COL_ROLE     = 15;
var COL_EXPECTED = 16;
var COL_REQID    = 17;

// O·P열 헤더가 없으면 만든다 (기존 행은 비워 둠)
function ensureRoleColumn_(sheet) {
  var h = String(sheet.getRange(1, COL_ROLE).getValue() || '').trim();
  if (h !== '참여형태') sheet.getRange(1, COL_ROLE).setValue('참여형태');
}
function ensureExpectedColumn_(sheet) {
  ensureColumn_(sheet, COL_EXPECTED, '예상금액');
}
function ensureReqIdColumn_(sheet) {
  ensureColumn_(sheet, COL_REQID, '요청ID');
}
function ensureColumn_(sheet, col, title) {
  if (sheet.getMaxColumns() < col) {
    sheet.insertColumnsAfter(sheet.getMaxColumns(), col - sheet.getMaxColumns());
  }
  var h = String(sheet.getRange(1, col).getValue() || '').trim();
  if (h !== title) sheet.getRange(1, col).setValue(title);
}

/* ── 같은 저장이 두 번 들어가는 것 방지 (멱등키) ──────────────
   네트워크가 흔들려 응답만 못 받으면 페이지가 다시 보낼 수 있다.
   그때 행이 두 번 생기지 않도록, 요청마다 붙어 오는 reqId 를 Q열에 남기고
   같은 reqId 가 이미 있으면 새로 만들지 않고 그때 만든 결과를 그대로 돌려준다.
   → 몇 번을 보내도 결과가 같다. 덕분에 add·complete 도 안전하게 재시도된다. */
function findRowByReqId_(sheet, reqId) {
  return findByReqIdInCol_(sheet, reqId, COL_REQID);
}
// 자격증현황 시트 I열(9) = 요청ID
var CERT_COL_REQID = 9;
function findCertByReqId_(sheet, reqId) {
  return findByReqIdInCol_(sheet, reqId, CERT_COL_REQID);
}
function findByReqIdInCol_(sheet, reqId, col) {
  if (!reqId) return -1;
  var last = sheet.getLastRow();
  if (last < 2 || sheet.getMaxColumns() < col) return -1;
  var vals = sheet.getRange(2, col, last - 1, 1).getValues();
  for (var i = 0; i < vals.length; i++) {
    if (String(vals[i][0] || '').trim() === String(reqId).trim()) return i + 2;
  }
  return -1;
}

// ============================================================
//  doGet — GET 요청 처리
//  ?type=schedule  → 강의 스케쥴 데이터 반환
//  (파라미터 없음)  → 기존 경력서 데이터 반환
// ============================================================
function doGet(e) {
  try {
    // 강의 스케쥴 요청인 경우
    if (e && e.parameter && e.parameter.type === 'schedule') {
      return getScheduleData_();
    }

    // 배포 버전 확인 — 파일이 없어도 죽지 않도록 typeof 로 감싼다
    if (e && e.parameter && e.parameter.type === 'version') {
      return jsonOutput({
        code:     CODE_VERSION,
        notion:   (typeof NOTION_VERSION   !== 'undefined') ? NOTION_VERSION   : '(notion.gs 없음)',
        calendar: (typeof CAL_VERSION      !== 'undefined') ? CAL_VERSION      : '(calendar.gs 없음)',
        at:       Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm')
      });
    }

    // 자동 검색 결과 요청
    if (e && e.parameter && e.parameter.sheet === '자동검색결과') {
      return getAutoSearchData_();
    }

    // 기존 경력서 요청 (변경 없음)
    var ss = getSS_();
    var result = {
      basicInfo:    BASIC_INFO,
      education:    readEducation(ss),
      career:       readCareer(ss),
      certificates: readCertificates(ss)
    };
    return jsonOutput(result);

  } catch (err) {
    return jsonOutput({ error: err.toString() });
  }
}

// ============================================================
//  doPost — POST 요청 처리
//  action: 'add'/'update'/'delete'  → 강의 스케쥴 처리 (새로 추가)
//  sheet: '알림요청'                 → 카카오 알림 (기존)
//  그 외                             → 시트 데이터 저장 (기존)
// ============================================================
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss   = getSS_();

    // 강의 스케쥴 처리 (action 키가 있는 경우)
    if (data.action === 'add' || data.action === 'update' || data.action === 'delete' || data.action === 'complete') {
      return handleSchedule_(ss, data);
    }

    // 노션 정산 원장 등록 (notion.gs)
    if (data.action === 'notionAdd') {
      return handleNotionAdd_(ss, data);
    }

    // 노션 정산 등록 — 입금 확인 후 여러 건을 금액까지 함께 기록 (notion.gs)
    if (data.action === 'notionSettle') {
      return handleNotionSettle_(ss, data);
    }

    // 노션 정산 원장 읽기 — 미수금 카드 · 월별 수입 요약 (notion.gs)
    if (data.action === 'notionLedger') {
      return handleNotionLedger_(data);
    }

    // 구글 캘린더 동기화 (calendar.gs)
    if (data.action === 'syncCalendar') {
      return handleCalendarSync_(data);
    }

    // 자격증 / 연수 처리
    if (data.action === 'certAdd' || data.action === 'certUpdate' || data.action === 'certDelete') {
      return handleCert_(ss, data);
    }

    // 자동 검색 결과 저장
    if (data.action === 'saveAutoSearch') {
      return saveAutoSearchData_(ss, data);
    }

    // 기존: 카카오 알림 요청
    if (data.sheet === '알림요청') {
      sendDailyKakaoNotification();
      return jsonOutput({ success: true, action: 'kakao_sent' });
    }

    // 기존: 그 외 시트 데이터 저장
    if (data.data && data.data.length > 0) {
      var sheet = ss.getSheetByName(data.sheet) || ss.insertSheet(data.sheet);
      sheet.clearContents();
      sheet.getRange(1, 1, data.data.length, data.data[0].length).setValues(data.data);
    }
    return jsonOutput({ success: true });

  } catch (err) {
    return jsonOutput({ error: err.toString() });
  }
}

// ============================================================
//  강의 스케쥴 — 읽기
//  시트 컬럼 순서: ID | 날짜 | 구분 | 업체명 | 수강방식 | 출강학교단체 | 시간 | 시수 | 수업주제 | 비고 | 경력등록(K)
//                 | 정산등록(L) | 캘린더ID(M) | 동기화서명(N) | 참여형태(O) | 예상금액(P)
// ============================================================
function getScheduleData_() {
  var ss    = getSS_();
  var sheet = ss.getSheetByName(SHEET_SCHEDULE);

  if (!sheet) {
    sheet = ss.insertSheet(SHEET_SCHEDULE);
    sheet.appendRow(['ID','날짜','구분','업체명','수강방식','출강학교단체','시간','시수','수업주제','비고','경력등록','정산등록','캘린더ID','동기화서명','참여형태','예상금액']);
    return jsonOutput([]);
  }

  var rows   = sheet.getDataRange().getValues();
  var result = [];
  for (var i = 1; i < rows.length; i++) {
    var r = rows[i];
    if (!r[0] && !r[1]) continue;
    // 날짜 셀이 Date 객체인 경우 yyyy-MM-dd 형식으로 변환
    var dateRaw = r[1];
    var dateStr = '';
    if (dateRaw instanceof Date) {
      dateStr = Utilities.formatDate(dateRaw, Session.getScriptTimeZone(), 'yyyy-MM-dd');
    } else {
      dateStr = String(dateRaw || '').trim();
    }
    result.push({
      id:        String(r[0]  || '').trim(),
      date:      dateStr,
      type:      String(r[2]  || '').trim(),
      company:   String(r[3]  || '').trim(),
      mode:      String(r[4]  || '').trim(),
      school:    String(r[5]  || '').trim(),
      time:      String(r[6]  || '').trim(),
      hours:     String(r[7]  || '').trim(),
      topic:     String(r[8]  || '').trim(),
      note:      String(r[9]  || '').trim(),
      completed: String(r[10] || '').trim(), // 경력등록 완료 여부 ('Y' or '')
      settled:   String(r[11] || '').trim(), // 정산등록(노션) 완료 여부
      role:      String(r[14] || '').trim(), // 참여형태 O열 (강사/보조강사/안전요원)
      expected:  String(r[15] || '').trim()  // 예상금액 P열 (노션 '예정' 행 금액)
    });
  }
  return jsonOutput(result);
}

// ============================================================
//  강의 스케쥴 — 추가 / 수정 / 삭제
// ============================================================
function handleSchedule_(ss, data) {
  var sheet = ss.getSheetByName(SHEET_SCHEDULE);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_SCHEDULE);
    sheet.appendRow(['ID','날짜','구분','업체명','수강방식','출강학교단체','시간','시수','수업주제','비고','경력등록','정산등록','캘린더ID','동기화서명','참여형태','예상금액']);
  }

  // 추가
  if (data.action === 'add') {
    // 이미 처리한 요청이면 그때 만든 강의를 그대로 돌려준다 (중복 등록 방지)
    var dup = findRowByReqId_(sheet, data.reqId);
    if (dup > 0) {
      return jsonOutput({ success: true, duplicate: true,
                          id: String(sheet.getRange(dup, 1).getValue() || '').trim() });
    }
    var newId = generateId_();
    sheet.appendRow([
      newId,
      String(data.date    || '').trim(),
      String(data.type    || '').trim(),
      String(data.company || '').trim(),
      String(data.mode    || '').trim(),
      String(data.school  || '').trim(),
      String(data.time    || '').trim(),
      String(data.hours   || '').trim(),
      String(data.topic   || '').trim(),
      String(data.note    || '').trim(),
      ''  // 경력등록 (초기값 빈 문자열)
    ]);
    // 참여형태(O열)·예상금액(P열)은 appendRow 범위 밖 컬럼이라 따로 기록한다
    var addedRow = sheet.getLastRow();
    if (data.role) {
      ensureRoleColumn_(sheet);
      sheet.getRange(addedRow, COL_ROLE).setValue(String(data.role).trim());
    }
    if (data.expected) {
      ensureExpectedColumn_(sheet);
      sheet.getRange(addedRow, COL_EXPECTED).setValue(String(data.expected).trim());
    }
    if (data.reqId) {
      ensureReqIdColumn_(sheet);
      sheet.getRange(addedRow, COL_REQID).setValue(String(data.reqId).trim());
    }
    // 예상금액이 있으면 노션에 '예정' 행을 만들어 월별 차트에 잡히게 한다
    syncNotionPlan_(sheet, addedRow);
    return jsonOutput({ success: true, id: newId });
  }

  // 수정
  if (data.action === 'update') {
    var rowIdx = findRowById_(sheet, data.id);
    if (rowIdx < 0) return errorOutput('해당 ID를 찾을 수 없습니다: ' + data.id);
    // 기존 경력등록 완료 플래그 보존
    var existingCompleted = '';
    try { existingCompleted = String(sheet.getRange(rowIdx, 11).getValue() || '').trim(); } catch(e) {}
    sheet.getRange(rowIdx, 1, 1, 11).setValues([[
      data.id,
      String(data.date    || '').trim(),
      String(data.type    || '').trim(),
      String(data.company || '').trim(),
      String(data.mode    || '').trim(),
      String(data.school  || '').trim(),
      String(data.time    || '').trim(),
      String(data.hours   || '').trim(),
      String(data.topic   || '').trim(),
      String(data.note    || '').trim(),
      existingCompleted
    ]]);
    // 참여형태(O)·예상금액(P) — 정산등록(L)·캘린더ID(M)·동기화서명(N)은 건드리지 않는다
    if (data.role) {
      ensureRoleColumn_(sheet);
      sheet.getRange(rowIdx, COL_ROLE).setValue(String(data.role).trim());
    }
    if (data.expected !== undefined) {
      ensureExpectedColumn_(sheet);
      sheet.getRange(rowIdx, COL_EXPECTED).setValue(String(data.expected || '').trim());
    }
    syncNotionPlan_(sheet, rowIdx);
    return jsonOutput({ success: true });
  }

  // 삭제 — 이미 지워졌으면 성공으로 본다 (재시도해도 오류가 나지 않게)
  if (data.action === 'delete') {
    var rowIdx = findRowById_(sheet, data.id);
    // 연동된 노션 행(입금완료 제외)도 함께 지운다 — notion.gs (재시도여도 안전)
    if (typeof notionRemoveLinked_ === 'function') notionRemoveLinked_(data.id, false);
    if (rowIdx < 0) return jsonOutput({ success: true, alreadyGone: true });
    sheet.deleteRow(rowIdx);
    return jsonOutput({ success: true });
  }

  // ── 경력 등록 완료 ──────────────────────────────────────────
  //  스케쥴 데이터를 경력사항 시트에 자동 입력하고 완료 플래그를 저장한다.
  // ────────────────────────────────────────────────────────────
  if (data.action === 'complete') {
    /* 이미 등록된 강의면 경력사항에 또 넣지 않는다.
       재시도로 같은 요청이 두 번 와도, 첫 요청이 K열을 'Y' 로 만들어 두므로 여기서 걸린다. */
    var doneIdx = findRowById_(sheet, data.id);
    if (doneIdx > 0 && String(sheet.getRange(doneIdx, 11).getValue() || '').trim() === 'Y') {
      return jsonOutput({ success: true, duplicate: true });
    }

    // 1. 경력사항 시트에 행 추가
    var careerSheet = ss.getSheetByName(SHEET_CAREER);
    if (!careerSheet) return errorOutput('경력사항 시트를 찾을 수 없습니다.');

    /* 가공 규칙은 서버가 갖는다(정산과 같은 방식).
       예전 페이지는 period/org/position/work 를 만들어 보내므로,
       값이 오면 그대로 쓰고 없을 때만 시트에서 만들어 낸다. */
    var c = buildCareerRow_(sheet, data);
    careerSheet.appendRow([c.period, c.org, c.position, c.work, c.note]);

    // 2. 강의스케쥴 시트에 완료 플래그('Y') 저장
    var rowIdx = findRowById_(sheet, data.id);
    if (rowIdx > 0) {
      // 헤더 행에 '경력등록' 컬럼이 없으면 추가
      var headerRow = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
      if (headerRow.length < 11 || headerRow[10] === '') {
        sheet.getRange(1, 11).setValue('경력등록');
      }
      sheet.getRange(rowIdx, 11).setValue('Y');
    }

    return jsonOutput({ success: true });
  }

  return errorOutput('알 수 없는 action: ' + data.action);
}

//  컬럼: A 카테고리 | B 명칭 | C 날짜 | D 발급/주관기관 | E 비고/결과 | F 급수/시간 | G 분류 | H ID

function handleCert_(ss, data) {
  var sheet = ss.getSheetByName(SHEET_CERT);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_CERT);
    sheet.appendRow(['카테고리','명칭','날짜','발급/주관기관','비고/결과','급수/시간','분류','ID']);
  }
  ensureCertHeader_(sheet);

  if (data.action === 'certAdd') {
    // 같은 요청이 두 번 오면(재시도) 그때 만든 것을 그대로 돌려준다
    var cdup = findCertByReqId_(sheet, data.reqId);
    if (cdup > 0) {
      return jsonOutput({ success: true, duplicate: true,
                          id: String(sheet.getRange(cdup, 8).getValue() || '').trim() });
    }
    var newId = generateId_();
    sheet.appendRow([
      String(data.category || '자격증').trim(),
      String(data.name   || '').trim(),
      String(data.date   || '').trim(),
      String(data.issuer || '').trim(),
      String(data.note   || '').trim(),
      String(data.extra  || '').trim(),
      String(data.group  || '').trim(),
      newId
    ]);
    if (data.reqId) {
      ensureColumn_(sheet, CERT_COL_REQID, '요청ID');
      sheet.getRange(sheet.getLastRow(), CERT_COL_REQID).setValue(String(data.reqId).trim());
    }
    return jsonOutput({ success: true, id: newId });
  }

  if (data.action === 'certUpdate') {
    var rowIdx = findRowByCol_(sheet, 8, data.id);
    if (rowIdx < 0) return errorOutput('해당 ID를 찾을 수 없습니다: ' + data.id);
    sheet.getRange(rowIdx, 1, 1, 8).setValues([[
      String(data.category || '자격증').trim(),
      String(data.name   || '').trim(),
      String(data.date   || '').trim(),
      String(data.issuer || '').trim(),
      String(data.note   || '').trim(),
      String(data.extra  || '').trim(),
      String(data.group  || '').trim(),
      data.id
    ]]);
    return jsonOutput({ success: true });
  }

  if (data.action === 'certDelete') {
    var rowIdx = findRowByCol_(sheet, 8, data.id);
    if (rowIdx < 0) return errorOutput('해당 ID를 찾을 수 없습니다: ' + data.id);
    sheet.deleteRow(rowIdx);
    return jsonOutput({ success: true });
  }

  return errorOutput('알 수 없는 cert action: ' + data.action);
}

/* 스케쥴 한 건 → 경력사항 행. 클라이언트가 이미 만들어 보낸 값이 있으면 그대로 쓴다.
   (예전 schedule.html 이 그렇게 보낸다 — 하위 호환)
   매핑: 날짜 2026-05-26 → 2026.05.26 / 교육수강·연수→수강 · 디지털튜터→디지털튜터 · 나머지→강사
         주요업무 = 출강학교 (시수h) — 수업주제                                        */
function buildCareerRow_(schedSheet, data) {
  var out = {
    period:   String(data.period   || '').trim(),
    org:      String(data.org      || '').trim(),
    position: String(data.position || '').trim(),
    work:     String(data.work     || '').trim(),
    note:     String(data.note     || '완료').trim()
  };
  if (out.period && out.org && out.position && out.work) return out;

  var rowIdx = findRowById_(schedSheet, data.id);
  if (rowIdx < 0) return out;
  var r = schedSheet.getRange(rowIdx, 1, 1, 11).getValues()[0];

  var dateRaw = r[1];
  var dateStr = (dateRaw instanceof Date)
    ? Utilities.formatDate(dateRaw, Session.getScriptTimeZone(), 'yyyy-MM-dd')
    : String(dateRaw || '').trim();

  var type   = String(r[2] || '').trim();
  var school = String(r[5] || '').trim();
  var hours  = String(r[7] || '').trim();
  var topic  = String(r[8] || '').trim();

  var hs   = hours ? ' (' + hours + 'h)' : '';
  var what = topic || (type + ' 강의');

  if (!out.period)   out.period   = dateStr.replace(/-/g, '.');
  if (!out.org)      out.org      = String(r[3] || '').trim();
  if (!out.position) out.position = (type === '교육수강' || type === '연수') ? '수강'
                                  : type === '디지털튜터' ? '디지털튜터'
                                  : '강사';
  if (!out.work)     out.work     = school ? (school + hs + ' — ' + what)
                                           : (what + hs);
  return out;
}

/* ============================================================
    경력 미등록 월간 알림 (매월 1일 오전 9시 메일)
    ------------------------------------------------------------
    경력등록은 사람이 직접 확인하고 누르는 게 맞다고 보아 자동 등록은 하지 않는다.
    대신 지난 강의 중 등록이 안 된 게 남아 있으면 한 달에 한 번 메일로 알린다.
    등록을 안 하면 프로필 페이지의 출강 학교 칩·통계에 반영되지 않는다.

    [ 설치 ] 편집기에서 setupCareerReminderTrigger 를 한 번 실행 → 권한 승인
    [ 테스트 ] monthlyCareerReminder 를 직접 실행하면 바로 메일이 온다
   ============================================================ */
var CAREER_ALERT_TO   = BASIC_INFO.email;   // 받는 사람
var CAREER_ALERT_PAGE = 'https://loveria3.github.io/profile/schedule.html';

// 날짜 파싱 (Date 또는 'yyyy-MM-dd')
function careerParseDate_(v) {
  if (v instanceof Date) {
    if (isNaN(v.getTime())) return null;
    return new Date(v.getFullYear(), v.getMonth(), v.getDate());
  }
  var m = String(v || '').trim().match(/^(\d{4})[-.\/](\d{1,2})[-.\/](\d{1,2})/);
  if (!m) return null;
  return new Date(+m[1], +m[2] - 1, +m[3]);
}

// 강의가 끝났는지 — 종료시각이 있으면 그 기준, 없으면 익일 0시 기준 (페이지와 같은 규칙)
function careerIsPast_(d, timeStr, now) {
  var m = String(timeStr || '').match(/[~\-–—](\d{1,2}):(\d{2})/);
  if (m) return now > new Date(d.getFullYear(), d.getMonth(), d.getDate(), +m[1], +m[2], 0);
  return now > new Date(d.getFullYear(), d.getMonth(), d.getDate() + 1, 0, 0, 0);
}

// 지난 강의 중 경력등록(K열)이 안 된 것
function pendingCareerRows_() {
  var sheet = getSS_().getSheetByName(SHEET_SCHEDULE);
  if (!sheet) return [];
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];

  var values = sheet.getRange(1, 1, lastRow, 11).getValues();
  var now = new Date(), out = [];
  for (var i = 1; i < values.length; i++) {
    var r = values[i];
    if (!String(r[0] || '').trim()) continue;
    if (String(r[10] || '').trim() === 'Y') continue;     // 이미 등록됨
    var d = careerParseDate_(r[1]);
    if (!d) continue;
    if (!careerIsPast_(d, String(r[6] || '').trim(), now)) continue;
    out.push({
      date:    Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd'),
      company: String(r[3] || '').trim(),
      school:  String(r[5] || '').trim(),
      topic:   String(r[8] || '').trim()
    });
  }
  out.sort(function (a, b) { return a.date < b.date ? -1 : a.date > b.date ? 1 : 0; });
  return out;
}

function monthlyCareerReminder() {
  var pend = pendingCareerRows_();
  if (!pend.length) {
    Logger.log('경력 미등록 없음 — 메일을 보내지 않습니다.');
    return 0;
  }

  var months = {}, lines = [];
  pend.forEach(function (p) {
    months[p.date.slice(0, 7)] = true;
    lines.push('  ' + p.date + '   ' + p.company
             + (p.school ? ' · ' + p.school : '')
             + (p.topic  ? ' · ' + p.topic  : ''));
  });
  var monthList = Object.keys(months).sort().map(function (m) {
    return m.slice(0, 4) + '년 ' + (+m.slice(5)) + '월';
  }).join(', ');

  var NL = String.fromCharCode(10);
  var body =
      '지난 강의 중 경력사항에 등록되지 않은 건이 ' + pend.length + '건 남아 있습니다.' + NL
    + '(' + monthList + ')' + NL + NL
    + lines.join(NL) + NL + NL
    + '등록해야 프로필 페이지의 출강 학교 칩과 통계에 반영됩니다.' + NL
    + '스케쥴 페이지에서 해당 월로 이동해 [경력등록] 또는 [경력등록 N건 일괄] 을 눌러주세요.' + NL + NL
    + CAREER_ALERT_PAGE + NL;

  MailApp.sendEmail({
    to:      CAREER_ALERT_TO,
    subject: '[강의 스케쥴] 경력 미등록 ' + pend.length + '건 확인 필요',
    body:    body
  });
  Logger.log('✅ 경력 미등록 ' + pend.length + '건 — ' + CAREER_ALERT_TO + ' 로 메일 발송');
  return pend.length;
}

function findRowById_(sheet, id) {
  var rows = sheet.getDataRange().getValues();
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][0]).trim() === String(id).trim()) return i + 1;
  }
  return -1;
}

// 지정한 컬럼(1-based)에서 ID 일치 행 찾기 (자격증현황은 ID가 8번째 컬럼)
function findRowByCol_(sheet, col, id) {
  var rows = sheet.getDataRange().getValues();
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][col - 1]).trim() === String(id).trim()) return i + 1;
  }
  return -1;
}

// 자격증현황 헤더에 새 컬럼(급수/시간·분류·ID)이 없으면 보강
function ensureCertHeader_(sheet) {
  var headers = ['카테고리','명칭','날짜','발급/주관기관','비고/결과','급수/시간','분류','ID'];
  var lastCol = sheet.getLastColumn();
  for (var c = lastCol; c < 8; c++) {
    sheet.getRange(1, c + 1).setValue(headers[c]);
  }
}

function refreshKakaoToken_() {
  var props = PropertiesService.getScriptProperties();
  var rt    = props.getProperty('kakao_refresh_token');
  if (!rt) { Logger.log('리프레시 토큰 없음'); return false; }
  var res  = UrlFetchApp.fetch('https://kauth.kakao.com/oauth/token', {
    method: 'post',
    payload: { grant_type: 'refresh_token', client_id: KAKAO_REST_KEY, refresh_token: rt }
  });
  var json = JSON.parse(res.getContentText());
  if (json.error) { Logger.log('갱신 오류: ' + json.error_description); return false; }
  PropertiesService.getScriptProperties().setProperty('kakao_access_token', json.access_token);
  if (json.refresh_token) PropertiesService.getScriptProperties().setProperty('kakao_refresh_token', json.refresh_token);
  return true;
}

function sendKakaoMessage_(text) {
  var props = PropertiesService.getScriptProperties();
  var token = props.getProperty('kakao_access_token');
  if (!token) { Logger.log('액세스 토큰 없음'); return false; }
  function doRequest_(t) {
    return UrlFetchApp.fetch('https://kapi.kakao.com/v2/api/talk/memo/default/send', {
      method: 'post',
      headers: { 'Authorization': 'Bearer ' + t },
      payload: { template_object: JSON.stringify({ object_type: 'text', text: text, link: { web_url: EDU_PAGE_URL || 'https://www.kakaocorp.com' } }) },
      muteHttpExceptions: true
    });
  }
  var res = doRequest_(token);
  var result = JSON.parse(res.getContentText());
  if (result.result_code === 0) return true;
  if (result.code === -401 && refreshKakaoToken_()) {
    token = PropertiesService.getScriptProperties().getProperty('kakao_access_token');
    res = doRequest_(token);
    result = JSON.parse(res.getContentText());
    return result.result_code === 0;
  }
  Logger.log('카카오 전송 실패: ' + JSON.stringify(result));
  return false;
}

// ============================================================
//  자동 검색 결과 — 읽기
//  GET ?sheet=자동검색결과 → { updatedAt, searchedAt, keywords, items[] }
// ============================================================
function getAutoSearchData_() {
  var ss    = getSS_();
  var sheet = ss.getSheetByName(SHEET_AUTO_SEARCH);

  // 시트가 없으면 빈 결과 반환
  if (!sheet || sheet.getLastRow() < 2) {
    return jsonOutput({ updatedAt: '', searchedAt: '', keywords: [], items: [] });
  }

  var rows   = sheet.getDataRange().getValues();
  // 1행: 메타 [updatedAt, searchedAt, keywords(JSON)]
  var meta   = rows[0];
  var result = {
    updatedAt:  String(meta[0] || ''),
    searchedAt: String(meta[1] || ''),
    keywords:   JSON.parse(meta[2] || '[]'),
    items:      []
  };

  // 3행~: 공고 데이터 [id, 제목, 대상, 필터, 일시, 장소, 주요내용, 신청링크, 상태]
  for (var i = 2; i < rows.length; i++) {
    var r = rows[i];
    if (!r[0]) continue;
    result.items.push({
      id:       String(r[0]),
      제목:     String(r[1] || ''),
      대상:     String(r[2] || ''),
      필터:     String(r[3] || ''),
      일시:     String(r[4] || ''),
      장소:     String(r[5] || ''),
      주요내용: String(r[6] || ''),
      신청링크: String(r[7] || ''),
      상태:     String(r[8] || '')
    });
  }
  return jsonOutput(result);
}

// ============================================================
//  자동 검색 결과 — 저장
//  POST action=saveAutoSearch, updatedAt, searchedAt, keywords[], items[]
// ============================================================
function saveAutoSearchData_(ss, data) {
  var sheet = ss.getSheetByName(SHEET_AUTO_SEARCH);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_AUTO_SEARCH);
  }
  sheet.clearContents();

  // 1행: 메타
  sheet.appendRow([
    data.updatedAt  || '',
    data.searchedAt || '',
    JSON.stringify(data.keywords || [])
  ]);
  // 2행: 헤더
  sheet.appendRow(['id','제목','대상','필터','일시','장소','주요내용','신청링크','상태']);
  // 3행~: 공고 데이터
  var items = data.items || [];
  items.forEach(function(item) {
    sheet.appendRow([
      item.id        || '',
      item.제목      || '',
      item.대상      || '',
      item.필터      || '',
      item.일시      || '',
      item.장소      || '',
      item.주요내용  || '',
      item.신청링크  || '',
      item.상태      || ''
    ]);
  });

  return jsonOutput({ success: true, saved: items.length });
}
